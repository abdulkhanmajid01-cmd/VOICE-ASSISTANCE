📋 CONFIGURATION QUICK REFERENCE
================================================================================

To customize, edit config.py:

# Wake words
WAKE_WORDS = ["jarvis", "assistant", "hey jarvis", "hey assistant"]

# GUI colors
GUI_THEME_COLOR = "#0D1117"         # Dark background
GUI_ACCENT_COLOR = "#58A6FF"        # Bright blue
GUI_SECONDARY_COLOR = "#1F6FEB"     # Medium blue

# Voice recognition
ENERGY_THRESHOLD = 2000             # Sensitivity (lower = more sensitive)
DYNAMIC_ENERGY_THRESHOLD = True     # Adaptive noise cancellation
LISTEN_TIMEOUT = 5                  # Seconds to wait for speech start

# Speech output
TTS_RATE = 150                      # Speed (100-200 recommended)
TTS_VOLUME = 0.9                    # Volume (0.0-1.0)

# Features
ENABLE_WAKE_WORD = True
ENABLE_BACKGROUND_LISTENING = True
ENABLE_GUI = True
